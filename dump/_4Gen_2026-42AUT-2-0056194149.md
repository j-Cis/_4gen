// src\layout\bar.rs

use crate::theme::{hue, ico};
use eframe::egui::{Align, CursorIcon, Image, Layout, Sense, Ui, Vec2};

#[derive(Clone)]
pub enum BarItem<'a> {
    Action {
        id: &'static str,
        icon: Image<'a>,
        tooltip: &'static str,
    },
    Separator,
}

pub struct Bar<'a> {
    pub vertical: bool,
    pub items: Vec<BarItem<'a>>,
}

impl<'a> Bar<'a> {
    pub fn vertical(items: Vec<BarItem<'a>>) -> Self {
        Self {
            vertical: true,
            items,
        }
    }

    pub fn horizontal(items: Vec<BarItem<'a>>) -> Self {
        Self {
            vertical: false,
            items,
        }
    }

    pub fn show(&self, ui: &mut Ui) -> Vec<&'static str> {
        let mut activated = Vec::new();

        ui.spacing_mut().item_spacing = egui::vec2(2.0, 2.0);

        if self.vertical {
            ui.with_layout(Layout::top_down(Align::Center), |ui| {
                for item in &self.items {
                    self.show_item(ui, item, &mut activated);
                }
            });
        } else {
            ui.with_layout(Layout::left_to_right(Align::Center), |ui| {
                for item in &self.items {
                    self.show_item(ui, item, &mut activated);
                }
            });
        }

        activated
    }

    fn show_item(&self, ui: &mut Ui, item: &BarItem, activated: &mut Vec<&'static str>) {
        match item {
            BarItem::Separator => {
                ui.add_space(4.0);
                ui.separator();
                ui.add_space(4.0);
            }
            BarItem::Action { id, icon, tooltip } => {
                let (rect, mut response) =
                    ui.allocate_exact_size(Vec2::splat(ico::ICON_BUTTON_SIZE), Sense::click());

                response = response.on_hover_cursor(CursorIcon::PointingHand);

                let bg_color = if response.is_pointer_button_down_on() {
                    hue::MAIN_TOOLBAR_ACTIVE
                } else if response.hovered() {
                    hue::MAIN_TOOLBAR_HOVER
                } else {
                    hue::MAIN_TOOLBAR_BG
                };

                ui.painter()
                    .rect_filled(rect, ico::ICON_BUTTON_CORNER_RADIUS, bg_color);
                let icon_rect =
                    eframe::egui::Rect::from_center_size(rect.center(), ico::ICON_SIZE_VEC);

                icon.clone()
                    .fit_to_exact_size(ico::ICON_SIZE_VEC)
                    .paint_at(ui, icon_rect);

                // Klasyczny tooltip dla prostych toolbarów
                if !tooltip.is_empty() {
                    response = response.on_hover_text(*tooltip);
                }

                if response.clicked() {
                    activated.push(*id);
                }
            }
        }
    }
}

use crate::theme::{hue, ico};
use eframe::egui::{self, Align, Align2, CursorIcon, Image, Layout, Sense, Ui, Vec2};

#[derive(Clone, Copy, PartialEq)]
pub enum FrameBarPlacement {
    Left,
    Right,
}

#[derive(Clone)]
pub enum FrameBarItem<'a> {
    Action {
        id: &'static str,
        icon: Image<'a>,
        tooltip: &'static str,
    },
    Separator,
}

pub struct FrameBar<'a> {
    pub placement: FrameBarPlacement,
    pub items: Vec<FrameBarItem<'a>>,
}

impl<'a> FrameBar<'a> {
    pub fn left(items: Vec<FrameBarItem<'a>>) -> Self {
        Self {
            placement: FrameBarPlacement::Left,
            items,
        }
    }

    pub fn right(items: Vec<FrameBarItem<'a>>) -> Self {
        Self {
            placement: FrameBarPlacement::Right,
            items,
        }
    }

    pub fn show(&self, ui: &mut Ui) -> Vec<&'static str> {
        let mut activated = Vec::new();

        ui.spacing_mut().item_spacing =
            egui::vec2(ico::ICON_PADDING_MAIN_AXIS, ico::ICON_PADDING_MAIN_AXIS);

        ui.with_layout(Layout::top_down(Align::Center), |ui| {
            for item in &self.items {
                self.show_item(ui, item, &mut activated);
            }
        });

        activated
    }

    fn show_item(&self, ui: &mut Ui, item: &FrameBarItem, activated: &mut Vec<&'static str>) {
        match item {
            FrameBarItem::Separator => {
                ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
                ui.separator();
                ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
            }
            FrameBarItem::Action { id, icon, tooltip } => {
                let (rect, mut response) =
                    ui.allocate_exact_size(Vec2::splat(ico::ICON_BUTTON_SIZE), Sense::click());

                response = response.on_hover_cursor(CursorIcon::PointingHand);

                let bg_color = if response.is_pointer_button_down_on() {
                    hue::MAIN_TOOLBAR_ACTIVE
                } else if response.hovered() {
                    hue::MAIN_TOOLBAR_HOVER
                } else {
                    hue::MAIN_TOOLBAR_BG
                };

                ui.painter()
                    .rect_filled(rect, ico::ICON_BUTTON_CORNER_RADIUS, bg_color);
                let icon_rect = egui::Rect::from_center_size(rect.center(), ico::ICON_SIZE_VEC);
                icon.clone()
                    .fit_to_exact_size(ico::ICON_SIZE_VEC)
                    .paint_at(ui, icon_rect);

                if response.hovered() && !tooltip.is_empty() {
                    let (tooltip_pos, tooltip_pivot) = match self.placement {
                        FrameBarPlacement::Left => (
                            egui::pos2(rect.right() + 6.0, rect.center().y),
                            Align2::LEFT_CENTER,
                        ),
                        FrameBarPlacement::Right => (
                            egui::pos2(rect.left() - 6.0, rect.center().y),
                            Align2::RIGHT_CENTER,
                        ),
                    };

                    egui::Area::new(ui.id().with(id).with("tooltip"))
                        .fixed_pos(tooltip_pos)
                        .pivot(tooltip_pivot)
                        .order(egui::Order::Tooltip)
                        .interactable(false)
                        .show(ui.ctx(), |ui| {
                            egui::Frame::popup(ui.style()).show(ui, |ui| {
                                ui.label(*tooltip);
                            });
                        });
                }

                if response.clicked() {
                    activated.push(*id);
                }
            }
        }
    }
}

use eframe::egui::{self, Color32, Rect, Sense, UiBuilder, vec2};
use crate::theme::tab::TableTheme;
use crate::layout::table_types::{PinPosition, SortDir, TableConfig, TableState};

pub struct TableGridEngine;

impl TableGridEngine {
    pub fn render<F>(
        ui: &mut egui::Ui,
        state: &mut TableState,
        config: &TableConfig,
        theme: &TableTheme,
        row_indices: &[usize],
        mut cell_renderer: F,
    ) where
        F: FnMut(usize, &str, &mut egui::Ui),
    {
        let active_order = state.get_computed_columns_order();

        let mut left_cols = Vec::new();
        let mut center_cols = Vec::new();

        for id in &active_order {
            if let Some(col) = state.columns.get(id) {
                match col.pinned {
                    PinPosition::Left => left_cols.push(id.clone()),
                    PinPosition::None => center_cols.push(id.clone()),
                }
            }
        }

        let shift_pressed = ui.input(|i| i.modifiers.shift);
        let ctrl_pressed = ui.input(|i| i.modifiers.command);

        let left_total_width: f32 = left_cols
            .iter()
            .map(|id| state.columns.get(id).map(|c| c.width_px).unwrap_or(100.0))
            .sum();

        let available_height = ui.available_height();

        // Identyfikator w pamięci egui do synchronizacji przesunięcia w poziomie (X)
        let scroll_x_id = ui.id().with("table_horizontal_scroll_sync");
        let mut scroll_x = ui.data(|d| d.get_temp::<f32>(scroll_x_id).unwrap_or(0.0));

        // ==========================================
        // 1. NAGŁÓWKI (ZAMROŻONE W PIONIE)
        // ==========================================
        ui.horizontal(|ui| {
            ui.spacing_mut().item_spacing.x = 0.0;

            if !left_cols.is_empty() {
                ui.allocate_ui_with_layout(
                    vec2(left_total_width, theme.header_height),
                    egui::Layout::left_to_right(egui::Align::Center),
                    |ui| {
                        Self::render_header_row(ui, &left_cols, state, theme, shift_pressed, true);
                    },
                );
                ui.separator();
            }

            if !center_cols.is_empty() {
                let header_scroll = egui::ScrollArea::horizontal()
                    .id_salt("header_hscroll")
                    .horizontal_scroll_offset(scroll_x)
                    .scroll_bar_visibility(egui::scroll_area::ScrollBarVisibility::AlwaysHidden)
                    .show(ui, |ui| {
                        Self::render_header_row(ui, &center_cols, state, theme, shift_pressed, false);
                    });

                // Aktualizujemy globalny offset jeśli nagłówek został przeciągnięty/przewinięty
                if header_scroll.state.offset.x != scroll_x {
                    scroll_x = header_scroll.state.offset.x;
                }
            }
        });

        // ==========================================
        // 2. CIAŁO TABELI (PRZEWIJANE W PIONIE)
        // ==========================================
        egui::ScrollArea::vertical()
            .id_salt("grid_vertical_scroll")
            .max_height(available_height - theme.header_height)
            .show(ui, |ui| {
                ui.horizontal(|ui| {
                    ui.spacing_mut().item_spacing.x = 0.0;

                    if !left_cols.is_empty() {
                        ui.allocate_ui_with_layout(
                            vec2(left_total_width, 0.0), // 0.0 = pozwól rosnąć do dołu
                            egui::Layout::top_down(egui::Align::LEFT),
                            |ui| {
                                Self::render_body_rows(
                                    ui, &left_cols, state, config, theme, row_indices, ctrl_pressed, &mut cell_renderer
                                );
                            },
                        );
                        ui.separator();
                    }

                    if !center_cols.is_empty() {
                        let body_scroll = egui::ScrollArea::horizontal()
                            .id_salt("body_hscroll")
                            .horizontal_scroll_offset(scroll_x)
                            .show(ui, |ui| {
                                ui.vertical(|ui| {
                                    Self::render_body_rows(
                                        ui, &center_cols, state, config, theme, row_indices, ctrl_pressed, &mut cell_renderer
                                    );
                                });
                            });

                        // Aktualizujemy globalny offset jeśli dolny pasek przewijania ruszył
                        if body_scroll.state.offset.x != scroll_x {
                            scroll_x = body_scroll.state.offset.x;
                        }
                    }
                });
            });

        // Zapisujemy nowy stan scroll_x dla następnej klatki
        ui.data_mut(|d| d.insert_temp(scroll_x_id, scroll_x));
    }

    fn render_header_row(
        ui: &mut egui::Ui,
        col_ids: &[String],
        state: &mut TableState,
        theme: &TableTheme,
        shift_pressed: bool,
        is_pinned: bool,
    ) {
        ui.horizontal(|ui| {
            ui.spacing_mut().item_spacing.x = 0.0;
            for col_id in col_ids {
                let (title, sortable, width_px) = match state.columns.get(col_id) {
                    Some(c) => (c.title.clone(), c.sortable, c.width_px),
                    None => continue,
                };

                let (rect, response) = ui.allocate_exact_size(
                    vec2(width_px, theme.header_height),
                    Sense::click_and_drag(),
                );

                ui.painter().rect_filled(rect, 0.0, theme.header_bg);

                let mut display_title = if is_pinned { format!("📌 {}", title) } else { title };
                if let Some((idx, (_, dir))) = state
                    .sort_rules
                    .iter()
                    .enumerate()
                    .find(|(_, (id, _))| id == col_id)
                {
                    let arrow = match dir {
                        SortDir::Asc => " 🔼",
                        SortDir::Desc => " 🔽",
                    };
                    let priority = if state.sort_rules.len() > 1 {
                        format!(" ({})", idx + 1)
                    } else {
                        "".to_string()
                    };
                    display_title = format!("{}{}{}", display_title, arrow, priority);
                }

                ui.painter().text(
                    rect.center(),
                    egui::Align2::CENTER_CENTER,
                    display_title,
                    egui::FontId::proportional(13.0),
                    Color32::WHITE,
                );

                if sortable && response.clicked() {
                    state.handle_sort_click(col_id, shift_pressed);
                }

                let resize_rect = Rect::from_min_max(
                    rect.right_top() - vec2(4.0, 0.0),
                    rect.right_bottom() + vec2(4.0, 0.0),
                );
                let resize_resp = ui.interact(
                    resize_rect,
                    ui.make_persistent_id(format!("resize_{}", col_id)),
                    Sense::drag(),
                );

                if resize_resp.dragged() {
                    let delta = resize_resp.drag_delta().x;
                    if let Some(col_def) = state.columns.get_mut(col_id) {
                        col_def.width_px = (col_def.width_px + delta).max(40.0);
                    }
                }

                if resize_resp.hovered() || resize_resp.dragged() {
                    ui.ctx().set_cursor_icon(egui::CursorIcon::ResizeColumn);
                }

                ui.painter().line_segment(
                    [rect.right_top(), rect.right_bottom()],
                    egui::Stroke::new(1.0, Color32::from_gray(60)),
                );
            }
        });
    }

    fn render_body_rows<F>(
        ui: &mut egui::Ui,
        col_ids: &[String],
        state: &mut TableState,
        config: &TableConfig,
        theme: &TableTheme,
        row_indices: &[usize],
        ctrl_pressed: bool,
        cell_renderer: &mut F,
    ) where
        F: FnMut(usize, &str, &mut egui::Ui),
    {
        for (display_idx, &true_idx) in row_indices.iter().enumerate() {
            let is_selected = state.selected_rows.contains(&true_idx);
            let row_bg = if is_selected {
                theme.row_selected_bg
            } else if display_idx % 2 == 0 {
                theme.row_even_bg
            } else {
                theme.row_odd_bg
            };

            ui.horizontal(|ui| {
                ui.spacing_mut().item_spacing.x = 0.0;
                for col_id in col_ids {
                    let width_px = state.columns.get(col_id).map(|c| c.width_px).unwrap_or(100.0);
                    let (rect, response) = ui.allocate_exact_size(
                        vec2(width_px, theme.row_height),
                        Sense::click(),
                    );

                    ui.painter().rect_filled(rect, 0.0, row_bg);

                    let mut cell_ui = ui.new_child(
                        UiBuilder::new()
                            .max_rect(rect.shrink(4.0))
                            .layout(egui::Layout::left_to_right(egui::Align::Center)),
                    );
                    cell_renderer(true_idx, col_id, &mut cell_ui);

                    if response.clicked() {
                        state.toggle_selection(true_idx, config, ctrl_pressed);
                    }

                    ui.painter().line_segment(
                        [rect.right_top(), rect.right_bottom()],
                        egui::Stroke::new(1.0, Color32::from_gray(35)),
                    );
                }
            });
        }
    }
}

pub struct TableState {
    pub columns: HashMap<String, ColDef>,
    pub display_order: Vec<String>,
    pub visible_columns: HashSet<String>,
    pub sort_rules: Vec<(String, SortDir)>,
    pub selected_rows: HashSet<usize>,

impl TableState {
    pub fn new(cols: Vec<ColDef>) -> Self {
        let mut columns = HashMap::new();
        let mut display_order = Vec::new();
        let mut visible_columns = HashSet::new();

        for col in cols {
            display_order.push(col.id.clone());
            visible_columns.insert(col.id.clone());
            columns.insert(col.id.clone(), col);
        }

        Self {
            columns,
            display_order,
            visible_columns,
            sort_rules: Vec::new(),
            selected_rows: HashSet::new(),
        }
    }

    pub fn get_computed_columns_order(&self) -> Vec<String> {
        let mut left_pinned = Vec::new();
        let mut unpinned = Vec::new();
        let mut right_pinned = Vec::new();

        for id in &self.display_order {
            if self.visible_columns.contains(id) {
                if let Some(col) = self.columns.get(id) {
                    match col.pinned {
                        PinPosition::Left => left_pinned.push(id.clone()),
                        PinPosition::None => unpinned.push(id.clone()),
                        // PinPosition::Right => right_pinned.push(id.clone()),
                    }
                }
            }
        }

        let mut final_order = left_pinned;
        final_order.extend(unpinned);
        final_order.extend(right_pinned);
        final_order
    }

    pub fn handle_sort_click(&mut self, col_id: &str, shift_pressed: bool) {
        if let Some(col) = self.columns.get(col_id) {
            if !col.sortable {
                return;
            }
        } else {
            return;
        }

        let existing_idx = self.sort_rules.iter().position(|(id, _)| id == col_id);

        if shift_pressed {
            if let Some(idx) = existing_idx {
                self.sort_rules[idx].1 = match self.sort_rules[idx].1 {
                    SortDir::Asc => SortDir::Desc,
                    SortDir::Desc => SortDir::Asc,
                };
            } else {
                self.sort_rules.push((col_id.to_string(), SortDir::Asc));
            }
        } else {
            let new_dir = if let Some(idx) = existing_idx {
                if self.sort_rules.len() == 1 {
                    match self.sort_rules[idx].1 {
                        SortDir::Asc => SortDir::Desc,
                        SortDir::Desc => SortDir::Asc,
                    }
                } else {
                    SortDir::Asc
                }
            } else {
                SortDir::Asc
            };
            self.sort_rules.clear();
            self.sort_rules.push((col_id.to_string(), new_dir));
        }
    }

    pub fn toggle_selection(&mut self, row_idx: usize, config: &TableConfig, ctrl_pressed: bool) {
        if !config.allow_selection {
            return;
        }

        if ctrl_pressed && config.multi_select {
            if self.selected_rows.contains(&row_idx) {
                self.selected_rows.remove(&row_idx);
            } else {
                if let Some(limit) = config.max_selection {
                    if self.selected_rows.len() >= limit {
                        return;
                    }
                }
                self.selected_rows.insert(row_idx);
            }
        } else {
            self.selected_rows.clear();
            self.selected_rows.insert(row_idx);
        }
    }
}

use egui::Color32;

/// Rozszerzenie Color32 o przestrzenie barw OKLab i OKLCH.
pub trait Color32OklabExt {
    fn from_oklab(l: f32, a: f32, b: f32) -> Self;
    fn from_oklch(l: f32, c: f32, h: f32) -> Self;
}

impl Color32OklabExt for Color32 {
    fn from_oklab(l: f32, a: f32, b: f32) -> Self {
        oklab(l, a, b)
    }

    fn from_oklch(l: f32, c: f32, h: f32) -> Self {
        oklch(l, c, h)
    }
}

/// Helper: Konwersja Oklab do liniowego sRGB (zwraca r, g, b bez obcinania)
const fn oklab_to_linear_srgb(l: f32, a: f32, b: f32) -> (f32, f32, f32) {
    let l_ = l + 0.3963377774 * a + 0.2158037573 * b;
    let m_ = l - 0.1055613458 * a - 0.0638541728 * b;
    let s_ = l - 0.0894841775 * a - 1.2914855480 * b;

    let ll = l_ * l_ * l_;
    let mm = m_ * m_ * m_;
    let ss = s_ * s_ * s_;

    let r = 4.0767416621 * ll - 3.3077115913 * mm + 0.2309699292 * ss;
    let g = -1.2684380046 * ll + 2.6097574011 * mm - 0.3413193965 * ss;
    let b = -0.0041960863 * ll - 0.7034186147 * mm + 1.7076147010 * ss;

    (r, g, b)
}

/// Helper: Sprawdza czy składowe RGB mieszczą się w gamucie sRGB (0.0 - 1.0)
const fn is_in_srgb_gamut(r: f32, g: f32, b: f32) -> bool {
    r >= 0.0 && r <= 1.0 && g >= 0.0 && g <= 1.0 && b >= 0.0 && b <= 1.0
}

/// Konwersja OKLab do sRGB — dostępna w czasie kompilacji.
pub const fn oklab(l: f32, a: f32, b: f32) -> Color32 {
    if l <= 0.0 {
        return Color32::BLACK;
    }
    let (r, g, b) = oklab_to_linear_srgb(l, a, b);
    Color32::from_rgb(channel(r), channel(g), channel(b))
}

/// Konwersja OKLCH (L, C, h w stopniach) do sRGB — const.
pub const fn oklch(l: f32, c: f32, h: f32) -> Color32 {
    if l <= 0.0 {
        return Color32::BLACK;
    }
    let l_clamped = if l < 0.0 {
        0.0
    } else if l > 1.0 {
        1.0
    } else {
        l
    };
    if c <= 0.0 {
        return oklab(l, 0.0, 0.0);
    }
    let (sin_h, cos_h) = sin_cos_degrees(h);

    let (mut r, mut g, mut b) = oklab_to_linear_srgb(l_clamped, c * cos_h, c * sin_h);

    if !is_in_srgb_gamut(r, g, b) {
        let mut min_c = 0.0;
        let mut max_c = c;
        let mut iters = 0;

        // BAZA: Rozpoczynamy od gwarantowanego koloru w gamucie (szarość dla danego L)
        let (mut best_r, mut best_g, mut best_b) = oklab_to_linear_srgb(l_clamped, 0.0, 0.0);

        while iters < 16 {
            let mid_c = min_c + (max_c - min_c) / 2.0;
            let (tr, tg, tb) = oklab_to_linear_srgb(l_clamped, mid_c * cos_h, mid_c * sin_h);

            if is_in_srgb_gamut(tr, tg, tb) {
                best_r = tr;
                best_g = tg;
                best_b = tb;
                min_c = mid_c;
            } else {
                max_c = mid_c;
            }
            iters += 1;
        }

        r = best_r;
        g = best_g;
        b = best_b;
    }

    Color32::from_rgb(channel(r), channel(g), channel(b))
}

const fn channel(linear: f32) -> u8 {
    let x = if linear < 0.0 {
        0.0
    } else if linear > 1.0 {
        1.0
    } else {
        linear
    };

    let srgb = if x <= 0.0031308 {
        12.92 * x
    } else {
        1.055 * pow_five_twelths(x) - 0.055
    };

    (srgb * 255.0 + 0.5) as u8
}

/// Oblicza x^(5/12) metodą Newtona:
/// Rozwiązujemy równanie y^12 = x, a następnie zwracamy y^5.
const fn pow_five_twelths(x: f32) -> f32 {
    let mut y = 1.0_f32;
    let mut i = 0;

    // Znajdujemy y = x^(1/12)
    while i < 16 {
        let y2 = y * y;
        let y4 = y2 * y2;
        let y8 = y4 * y4;
        let y11 = y8 * y2 * y;
        let y12 = y11 * y;

        y = y - (y12 - x) / (12.0 * y11);
        i += 1;
    }

    // Podnosimy wynik do potęgi 5, otrzymując x^(5/12)
    let y2 = y * y;
    let y4 = y2 * y2;
    y4 * y
}

/// Sinus i cosinus kąta w stopniach.
const fn sin_cos_degrees(degrees: f32) -> (f32, f32) {
    let mut angle = degrees;

    angle = angle % 360.0;
    if angle < 0.0 {
        angle += 360.0;
    }
    if angle > 180.0 {
        angle -= 360.0;
    }

    let mut x = angle * 0.017453292519943295;
    let mut cos_sign = 1.0_f32;

    if x > 1.5707963267948966 {
        x = 3.141592653589793 - x;
        cos_sign = -1.0;
    } else if x < -1.5707963267948966 {
        x = -3.141592653589793 - x;
        cos_sign = -1.0;
    }

    let x2 = x * x;

    let sin = x
        * (1.0
            + x2 * (-1.0 / 6.0
                + x2 * (1.0 / 120.0
                    + x2 * (-1.0 / 5040.0
                        + x2 * (1.0 / 362880.0
                            + x2 * (-1.0 / 39916800.0 + x2 * (1.0 / 6227020800.0)))))));
    let cos = 1.0
        + x2 * (-1.0 / 2.0
            + x2 * (1.0 / 24.0
                + x2 * (-1.0 / 720.0
                    + x2 * (1.0 / 40320.0
                        + x2 * (-1.0 / 3628800.0
                            + x2 * (1.0 / 479001600.0 + x2 * (-1.0 / 87178291200.0)))))));

    (sin, cos * cos_sign)
}

                std::process::exit(1);
            }
            let mut opts = history.clone();
            opts.push("< back".to_string());
            let choice = Select::new("Wybierz polecenie z historii:", opts)
                .prompt()
                .unwrap_or_else(|_| std::process::exit(0));

                return;

            }
        }

        save_to_history(&command_id, history);

fn read_history() -> Vec<String> {
    fs::read_to_string(HISTORY_FILE)
        .unwrap_or_default()
        .lines()
        .filter(|s| !s.trim().is_empty())
        .map(|s| s.to_string())
        .collect()
}

fn save_to_history(cmd: &str, mut history: Vec<String>) {
    history.retain(|x| x != cmd);
    history.insert(0, cmd.to_string());
    history.truncate(10);
    let _ = fs::write(HISTORY_FILE, history.join("\n"));
}
