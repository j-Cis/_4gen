pub struct ViewRegistry {
    views: HashMap<&'static str, Box<dyn CanvasView>>,
}

impl ViewRegistry {
    pub fn new() -> Self {
        Self {
            views: HashMap::new(),
        }
    }

    pub fn register(&mut self, id: &'static str, view: Box<dyn CanvasView>) {
        self.views.insert(id, view);
    }

    pub fn title(&self, id: &str) -> Option<&'static str> {
        self.views.get(id).map(|v| v.title())
    }

    pub fn show(&mut self, id: &str, ui: &mut Ui) {
        if let Some(view) = self.views.get_mut(id) {
            view.show(ui);

    pub id: String,
    pub title: String,
    pub icon: String,
    pub mode: ContainerWindowMode,
    pub is_open: bool,

    pub is_movable: bool,

impl ContainerWindow {
    pub fn new(
        id: impl Into<String>,
        title: impl Into<String>,
        icon: impl Into<String>,
        mode: ContainerWindowMode,
    ) -> Self {
        Self {
            id: id.into(),
            title: title.into(),
            icon: icon.into(),
            mode,
            is_open: false,
            is_minimized: false,
            is_movable: true,
            is_resizable: true,
            default_pos: None,
            default_size: Some(default_window_size()),
            min_size: default_min_window_size(),
        }
    }

        self.is_open = true;
    }

    pub fn close(&mut self) {
        self.is_open = false;
    }

    pub fn toggle(&mut self) {
        self.is_open = !self.is_open;
    }

    pub fn with_pos(mut self, pos: Pos2) -> Self {
        self.default_pos = Some(pos);
        self
    }

    pub fn with_size(mut self, size: Vec2) -> Self {
        self.default_size = Some(size);
        self
    }

        if self.icon.is_empty() {
            self.title.clone()
        } else {
            format!("{} {}", self.icon, self.title)
        }
    }

    where
        F: FnOnce(&mut egui::Ui),
    {
        if !self.is_open {
            return;
        }
        render_container_window(self, ui, add_contents);
    }

pub fn render_container_window<F>(window: &mut ContainerWindow, ui: &mut egui::Ui, add_contents: F)
where
    F: FnOnce(&mut egui::Ui),
{
    let bounds = ui.max_rect();
    let screen_rect = ui
        .ctx()
        .input(|i| i.raw.screen_rect.unwrap_or(egui::Rect::EVERYTHING));

    let header_config = build_window_header();
    let title = window.formatted_title();
    let is_movable = window.is_movable;
    let is_minimized = window.is_minimized;

    let mut close_requested = false;
    let mut toggle_minimize = false;

    match window.mode {
        ContainerWindowMode::Floating => {
            let pos_id = Id::new(&window.id).with("float_pos");
            let mut current_pos = ui.data(|d| d.get_temp::<Pos2>(pos_id)).unwrap_or_else(|| {
                window.default_pos.unwrap_or(bounds.center() - Vec2::new(180.0, 120.0))
            });

            let window_id = Id::new(&window.id);
            let mut drag_delta = Vec2::ZERO;

            let min_size = if is_minimized {
                Vec2::new(window.min_size.x, 0.0)
            } else {
                window.min_size
            };

            let mut win = egui::Window::new(&title)
                .id(window_id)
                .constrain_to(bounds)
                .title_bar(false)
                .resizable(window.is_resizable && !is_minimized)
                .movable(false)
                .frame(Frame::window(ui.style()).fill(hue::WINDOW_CONTENT_BG).inner_margin(0))
                .min_size(min_size);

            if let Some(pos) = window.default_pos {
                win = win.default_pos(pos);
            }
            if let Some(sz) = window.default_size {
                win = win.default_size(sz);
            }

            win.current_pos(current_pos).show(ui.ctx(), |window_ui| {
                let action = draw_custom_header(
                    window_ui,
                    window_id,
                    &title,
                    is_movable,
                    &header_config,
                );
                drag_delta = action.drag_delta;
                if action.close_clicked {
                    close_requested = true;
                }
                if action.toggle_minimize {
                    toggle_minimize = true;
                }

                if !is_minimized {
                    window_ui.add_space(4.0);
                    Frame::new().inner_margin(8).show(window_ui, |content_ui| {
                        add_contents(content_ui);
                    });
                }
            });

            if close_requested {
                window.is_open = false;
            }
            if toggle_minimize {

#[derive(Clone, Copy, PartialEq)]
pub enum SideNavPlacement {
    Left,
    Right,

#[derive(Clone)]
pub enum SideNavItem<'a> {
    Action {
        id: &'static str,
        icon: Image<'a>,
        tooltip: &'static str,
    },
    Separator,
}

pub struct SideNav<'a> {
    pub placement: SideNavPlacement,
    pub items: Vec<SideNavItem<'a>>,
}

impl<'a> SideNav<'a> {
    pub fn left(items: Vec<SideNavItem<'a>>) -> Self {
        Self {
            placement: SideNavPlacement::Left,
            items,
        }
    }

    pub fn right(items: Vec<SideNavItem<'a>>) -> Self {
        Self {
            placement: SideNavPlacement::Right,
            items,
        }
    }

    pub fn show(&self, ui: &mut Ui) -> Vec<&'static str> {
        let mut activated = Vec::new();

        ui.spacing_mut().item_spacing =
            egui::vec2(ico::ICON_PADDING_MAIN_AXIS, ico::ICON_PADDING_MAIN_AXIS);

        ui.with_layout(Layout::top_down(Align::Center), |ui| {
            for item in &self.items {
                self.show_item(ui, item, &mut activated);
            }
        });

        activated
    }

    fn show_item(&self, ui: &mut Ui, item: &SideNavItem, activated: &mut Vec<&'static str>) {
        match item {
            SideNavItem::Separator => {
                ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
                ui.separator();
                ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
            }
            SideNavItem::Action { id, icon, tooltip } => {
                let (rect, mut response) =
                    ui.allocate_exact_size(Vec2::splat(ico::ICON_BUTTON_SIZE), Sense::click());

                response = response.on_hover_cursor(CursorIcon::PointingHand);

                let bg_color = if response.is_pointer_button_down_on() {
                    hue::MAIN_TOOLBAR_ACTIVE
                } else if response.hovered() {
                    hue::MAIN_TOOLBAR_HOVER
                } else {
                    hue::MAIN_TOOLBAR_BG
                };

                ui.painter()
                    .rect_filled(rect, ico::ICON_BUTTON_CORNER_RADIUS, bg_color);
                let icon_rect = egui::Rect::from_center_size(rect.center(), ico::ICON_SIZE_VEC);
                icon.clone()
                    .fit_to_exact_size(ico::ICON_SIZE_VEC)
                    .paint_at(ui, icon_rect);

                if response.hovered() && !tooltip.is_empty() {
                    let (tooltip_pos, tooltip_pivot) = match self.placement {
                        SideNavPlacement::Left => (
                            egui::pos2(rect.right() + 6.0, rect.center().y),
                            Align2::LEFT_CENTER,
                        ),
                        SideNavPlacement::Right => (
                            egui::pos2(rect.left() - 6.0, rect.center().y),
                            Align2::RIGHT_CENTER,
                        ),
                    };

                    egui::Area::new(ui.id().with(id).with("tooltip"))
                        .fixed_pos(tooltip_pos)
                        .pivot(tooltip_pivot)
                        .order(egui::Order::Tooltip)
                        .interactable(false)
                        .show(ui.ctx(), |ui| {
                            egui::Frame::popup(ui.style()).show(ui, |ui| {
                                ui.label(*tooltip);
                            });
                        });
                }

                if response.clicked() {
                    activated.push(*id);
                }
            }
        }
    }

                    PinPosition::Left => left_cols.push(id.clone()),
                    PinPosition::None => center_cols.push(id.clone()),
                }
            }

        let ctrl_pressed = ui.input(|i| state.is_ctrl_active(i));

            .iter()
            .map(|id| state.columns.get(id).map(|c| c.width_px).unwrap_or(100.0))
            .sum();

        let available_height = ui.available_height();

        // Identyfikator w pamięci egui do synchronizacji przesunięcia w poziomie (X)
        let scroll_x_id = ui.id().with("table_horizontal_scroll_sync");
        let mut scroll_x = ui.data(|d| d.get_temp::<f32>(scroll_x_id).unwrap_or(0.0));

        // ==========================================
        // 1. NAGŁÓWKI (ZAMROŻONE W PIONIE)

            ui.spacing_mut().item_spacing.x = 0.0;

                ui.allocate_ui_with_layout(
                    vec2(left_total_width, theme.header_height),
                    egui::Layout::left_to_right(egui::Align::Center),
                    |ui| {
                        header::render_header_row(ui, &left_cols, state, theme, shift_pressed, true);

                        header::render_header_row(ui, &center_cols, state, theme, shift_pressed, false);
                    });

                // Aktualizujemy globalny offset jeśli nagłówek został przeciągnięty/przewinięty

                    scroll_x = header_scroll.state.offset.x;
                }
            }
        });

        // ==========================================

            .id_salt("grid_vertical_scroll")
            .max_height(available_height - theme.header_height)

                    ui.spacing_mut().item_spacing.x = 0.0;

                });
            });

pub fn render_header_row(
    ui: &mut egui::Ui,
    col_ids: &[String],
    state: &mut TableState,
    theme: &TableTheme,
    shift_pressed: bool,
    is_pinned: bool,
) {
    ui.horizontal(|ui| {
        ui.spacing_mut().item_spacing.x = 0.0;
        for col_id in col_ids {
            let (title, sortable, width_px) = match state.columns.get(col_id) {
                Some(c) => (c.title.clone(), c.sortable, c.width_px),
                None => continue,
            };

            let (rect, response) = ui.allocate_exact_size(
                vec2(width_px, theme.header_height),
                Sense::click_and_drag(),
            );

            ui.painter().rect_filled(rect, 0.0, theme.header_bg);

            let mut display_title = if is_pinned { format!("📌 {}", title) } else { title };
            if let Some((idx, (_, dir))) = state
                .sort_rules
                .iter()
                .enumerate()
                .find(|(_, (id, _))| id == col_id)
            {
                let arrow = match dir {
                    SortDir::Asc => " 🔼",
                    SortDir::Desc => " 🔽",
                };
                let priority = if state.sort_rules.len() > 1 {
                    format!(" ({})", idx + 1)
                } else {
                    "".to_string()
                };
                display_title = format!("{}{}{}", display_title, arrow, priority);
            }

            ui.painter().text(
                rect.center(),
                egui::Align2::CENTER_CENTER,
                display_title,
                egui::FontId::proportional(13.0),
                Color32::WHITE,
            );

            if sortable && response.clicked() {
                state.handle_sort_click(col_id, shift_pressed);
            }

            let resize_rect = Rect::from_min_max(
                rect.right_top() - vec2(4.0, 0.0),
                rect.right_bottom() + vec2(4.0, 0.0),
            );
            let resize_resp = ui.interact(
                resize_rect,
                ui.make_persistent_id(format!("resize_{}", col_id)),
                Sense::drag(),
            );

            if resize_resp.dragged() {
                let delta = resize_resp.drag_delta().x;
                if let Some(col_def) = state.columns.get_mut(col_id) {
                    col_def.width_px = (col_def.width_px + delta).max(40.0);
                }
            }

            if resize_resp.hovered() || resize_resp.dragged() {
                ui.ctx().set_cursor_icon(egui::CursorIcon::ResizeColumn);
            }

            ui.painter().line_segment(
                [rect.right_top(), rect.right_bottom()],
                egui::Stroke::new(1.0, Color32::from_gray(60)),
            );
        }
    });
}

    pub fn new(cols: Vec<ColDef>) -> Self {

        let mut display_order = Vec::new();
        let mut visible_columns = HashSet::new();

        for col in cols {
            display_order.push(col.id.clone());
            visible_columns.insert(col.id.clone());
            columns.insert(col.id.clone(), col);
        }

    pub fn is_ctrl_active(&self, input: &eframe::egui::InputState) -> bool {
        self.simulated_ctrl || input.modifiers.command || input.modifiers.ctrl
    }

    pub fn is_alt_active(&self, input: &eframe::egui::InputState) -> bool {
        self.simulated_alt || input.modifiers.alt
    }

    pub fn get_computed_columns_order(&self) -> Vec<String> {
        let mut left_pinned = Vec::new();
        let mut unpinned = Vec::new();
        let right_pinned = Vec::new();

        for id in &self.display_order {
            if self.visible_columns.contains(id) {
                if let Some(col) = self.columns.get(id) {
                    match col.pinned {
                        PinPosition::Left => left_pinned.push(id.clone()),
                        PinPosition::None => unpinned.push(id.clone()),
                        // PinPosition::Right => right_pinned.push(id.clone()),
                    }
                }
            }
        }

        let mut final_order = left_pinned;
        final_order.extend(unpinned);
        final_order.extend(right_pinned);
        final_order
    }

    pub fn handle_sort_click(&mut self, col_id: &str, shift_pressed: bool) {
        if let Some(col) = self.columns.get(col_id) {
            if !col.sortable {
                return;
            }
        } else {
            return;
        }

        let existing_idx = self.sort_rules.iter().position(|(id, _)| id == col_id);

        if shift_pressed {
            if let Some(idx) = existing_idx {
                self.sort_rules[idx].1 = match self.sort_rules[idx].1 {
                    SortDir::Asc => SortDir::Desc,
                    SortDir::Desc => SortDir::Asc,
                };
            } else {
                self.sort_rules.push((col_id.to_string(), SortDir::Asc));
            }
        } else {
            let new_dir = if let Some(idx) = existing_idx {
                if self.sort_rules.len() == 1 {
                    match self.sort_rules[idx].1 {
                        SortDir::Asc => SortDir::Desc,
                        SortDir::Desc => SortDir::Asc,
                    }
                } else {
                    SortDir::Asc
                }
            } else {
                SortDir::Asc
            };
            self.sort_rules.clear();
            self.sort_rules.push((col_id.to_string(), new_dir));
        }
    }

    pub fn toggle_selection(&mut self, row_idx: usize, config: &TableConfig, ctrl_pressed: bool) {
        if !config.allow_selection {
            return;
        }

        if ctrl_pressed && config.multi_select {
            if self.selected_rows.contains(&row_idx) {
                self.selected_rows.remove(&row_idx);
            } else {
                if let Some(limit) = config.max_selection {
                    if self.selected_rows.len() >= limit {
                        return;
                    }
                }
                self.selected_rows.insert(row_idx);
            }
        } else {
            self.selected_rows.clear();
            self.selected_rows.insert(row_idx);
        }
    }
}

use egui::Color32;

/// Rozszerzenie Color32 o przestrzenie barw OKLab i OKLCH.
pub trait Color32OklabExt {
    fn from_oklab(l: f32, a: f32, b: f32) -> Self;
    fn from_oklch(l: f32, c: f32, h: f32) -> Self;
}

impl Color32OklabExt for Color32 {
    fn from_oklab(l: f32, a: f32, b: f32) -> Self {
        oklab(l, a, b)
    }

    fn from_oklch(l: f32, c: f32, h: f32) -> Self {
        oklch(l, c, h)
    }
}

/// Helper: Konwersja Oklab do liniowego sRGB (zwraca r, g, b bez obcinania)
const fn oklab_to_linear_srgb(l: f32, a: f32, b: f32) -> (f32, f32, f32) {
    let l_ = l + 0.3963377774 * a + 0.2158037573 * b;
    let m_ = l - 0.1055613458 * a - 0.0638541728 * b;
    let s_ = l - 0.0894841775 * a - 1.2914855480 * b;

    let ll = l_ * l_ * l_;
    let mm = m_ * m_ * m_;
    let ss = s_ * s_ * s_;

    let r = 4.0767416621 * ll - 3.3077115913 * mm + 0.2309699292 * ss;
    let g = -1.2684380046 * ll + 2.6097574011 * mm - 0.3413193965 * ss;
    let b = -0.0041960863 * ll - 0.7034186147 * mm + 1.7076147010 * ss;

    (r, g, b)
}

/// Helper: Sprawdza czy składowe RGB mieszczą się w gamucie sRGB (0.0 - 1.0)
const fn is_in_srgb_gamut(r: f32, g: f32, b: f32) -> bool {
    r >= 0.0 && r <= 1.0 && g >= 0.0 && g <= 1.0 && b >= 0.0 && b <= 1.0
}

/// Konwersja OKLab do sRGB — dostępna w czasie kompilacji.
pub const fn oklab(l: f32, a: f32, b: f32) -> Color32 {
    if l <= 0.0 {
        return Color32::BLACK;
    }
    let (r, g, b) = oklab_to_linear_srgb(l, a, b);
    Color32::from_rgb(channel(r), channel(g), channel(b))
}

/// Konwersja OKLCH (L, C, h w stopniach) do sRGB — const.
pub const fn oklch(l: f32, c: f32, h: f32) -> Color32 {
    if l <= 0.0 {
        return Color32::BLACK;
    }
    let l_clamped = if l < 0.0 {
        0.0
    } else if l > 1.0 {
        1.0
    } else {
        l
    };
    if c <= 0.0 {
        return oklab(l, 0.0, 0.0);
    }
    let (sin_h, cos_h) = sin_cos_degrees(h);

    let (mut r, mut g, mut b) = oklab_to_linear_srgb(l_clamped, c * cos_h, c * sin_h);

    if !is_in_srgb_gamut(r, g, b) {
        let mut min_c = 0.0;
        let mut max_c = c;
        let mut iters = 0;

        // BAZA: Rozpoczynamy od gwarantowanego koloru w gamucie (szarość dla danego L)
        let (mut best_r, mut best_g, mut best_b) = oklab_to_linear_srgb(l_clamped, 0.0, 0.0);

        while iters < 16 {
            let mid_c = min_c + (max_c - min_c) / 2.0;
            let (tr, tg, tb) = oklab_to_linear_srgb(l_clamped, mid_c * cos_h, mid_c * sin_h);

            if is_in_srgb_gamut(tr, tg, tb) {
                best_r = tr;
                best_g = tg;
                best_b = tb;
                min_c = mid_c;
            } else {
                max_c = mid_c;
            }
            iters += 1;
        }

        r = best_r;
        g = best_g;
        b = best_b;
    }

    Color32::from_rgb(channel(r), channel(g), channel(b))
}

const fn channel(linear: f32) -> u8 {
    let x = if linear < 0.0 {
        0.0
    } else if linear > 1.0 {
        1.0
    } else {
        linear
    };

    let srgb = if x <= 0.0031308 {
        12.92 * x
    } else {
        1.055 * pow_five_twelths(x) - 0.055
    };

    (srgb * 255.0 + 0.5) as u8
}

/// Oblicza x^(5/12) metodą Newtona:
/// Rozwiązujemy równanie y^12 = x, a następnie zwracamy y^5.
const fn pow_five_twelths(x: f32) -> f32 {
    let mut y = 1.0_f32;
    let mut i = 0;

    // Znajdujemy y = x^(1/12)
    while i < 16 {
        let y2 = y * y;
        let y4 = y2 * y2;
        let y8 = y4 * y4;
        let y11 = y8 * y2 * y;
        let y12 = y11 * y;

        y = y - (y12 - x) / (12.0 * y11);
        i += 1;
    }

    // Podnosimy wynik do potęgi 5, otrzymując x^(5/12)
    let y2 = y * y;
    let y4 = y2 * y2;
    y4 * y
}

/// Sinus i cosinus kąta w stopniach.
const fn sin_cos_degrees(degrees: f32) -> (f32, f32) {
    let mut angle = degrees;

    angle = angle % 360.0;
    if angle < 0.0 {
        angle += 360.0;
    }
    if angle > 180.0 {
        angle -= 360.0;
    }

    let mut x = angle * 0.017453292519943295;
    let mut cos_sign = 1.0_f32;

    if x > 1.5707963267948966 {
        x = 3.141592653589793 - x;
        cos_sign = -1.0;
    } else if x < -1.5707963267948966 {
        x = -3.141592653589793 - x;
        cos_sign = -1.0;
    }

    let x2 = x * x;

    let sin = x
        * (1.0
            + x2 * (-1.0 / 6.0
                + x2 * (1.0 / 120.0
                    + x2 * (-1.0 / 5040.0
                        + x2 * (1.0 / 362880.0
                            + x2 * (-1.0 / 39916800.0 + x2 * (1.0 / 6227020800.0)))))));
    let cos = 1.0
        + x2 * (-1.0 / 2.0
            + x2 * (1.0 / 24.0
                + x2 * (-1.0 / 720.0
                    + x2 * (1.0 / 40320.0
                        + x2 * (-1.0 / 3628800.0
                            + x2 * (1.0 / 479001600.0 + x2 * (-1.0 / 87178291200.0)))))));

    (sin, cos * cos_sign)
}

            }
            let mut opts = history.clone();
            opts.push("< back".to_string());
            let choice = Select::new("Wybierz polecenie z historii:", opts)
                .prompt()
                .unwrap_or_else(|_| std::process::exit(0));
            
            if choice == "< back" {
                return;
            }
            choice
        }
        _ => {
            println!("\n🚀 4gen TUI");
            match run_interactive_menu(&history) {
                Some(id) => id,

            }
        }
    };

    if command_id != "clean" && command_id != "info-ip" && !command_id.starts_with("git-") {
        save_to_history(&command_id, history);

    execute_task(&command_id);

fn read_history() -> Vec<String> {
    fs::read_to_string(HISTORY_FILE)
        .unwrap_or_default()
        .lines()
        .filter(|s| !s.trim().is_empty())
        .map(|s| s.to_string())
        .collect()
}

fn save_to_history(cmd: &str, mut history: Vec<String>) {
    history.retain(|x| x != cmd);
    history.insert(0, cmd.to_string());
    history.truncate(10);
    let _ = fs::write(HISTORY_FILE, history.join("\n"));
}

use querypath::QueryPath;
use querypath_fmt::{
	CasePrecedence, CharClass, Column, DirWeightDisplay, GroupStrategy, NoExtPriority, NodeGroup,
	QueryPathBundle, QueryPathFmt, SameNamePriority, Sorting, StatsTemporal, StatsWeight,
	UnitSystem, WeightPrecision,
};
use querypath_snapshot::Snapshot;

pub fn generate_snapshot() {
	println!("🔍");
	// 1. Skanowanie systemu plików
	let results = QueryPath::new()
		.scan_at(["./"])
		.keep_parent(true)
		.match_pattern([
			"./{examples|tests|src|scripts|.github|.xtask|.cargo}/**",
			"./{Cargo|Trunk}.toml",
			"./index.html",
			"./assets/**",
			"./{.gitattributes|.gitignore}",
			"!./.xtask/target/**",
			"!./.xtask/Cargo.lock",
			"!./Cargo.lock",
			"!./{target|.git|dist|dump}/**",
		])
		.run()
		.expect("Błąd skanowania");

	// 2. Strategia sortowania
	let sorting = Sorting::new()
		.enabled(true)
		.group_strategy(GroupStrategy::Custom(vec![
			NodeGroup::TextFile,
			NodeGroup::Directory,
			NodeGroup::BinaryFile,
		]))
		.same_name_priority(SameNamePriority::FileFirst)
		.no_ext_priority(NoExtPriority::Above)
		.ignore_leading_dot(true)
		.char_class_order([CharClass::Special, CharClass::Digit, CharClass::Letter])
		.case_precedence(CasePrecedence::Insensitive);

	// 3. Złożenie, posortowanie i nadanie atomowej numeracji w drzewie
	let idn = QueryPathBundle::new()
		.queueing_before_numerate(sorting)
		.numerate_dirs(false)
		.numerate_binaries(false)
		.numerate_skip(["./assets/**", "*.svg"])
		.numerate_start_from(1)
		.apply_to(&results);

	// 4. Konfiguracja widoku kolumn i stylów drzewa
	let fmt = QueryPathFmt::new()
		.name_width(25)
		.path_width(35)
		.column_order_left(vec![])
		.column_order_right(vec![Column::Weight, Column::Temporal, Column::Path])
		.stats_weight(
			StatsWeight::new()
				.enabled(true)
				.unit_system(UnitSystem::Binary)
				.dir_display(DirWeightDisplay::MatchedOnly)
				.precision(WeightPrecision::Tenths),
		)
		.stats_temporal(StatsTemporal::new().enabled(true).pattern("WYYY-WW-D hh:mm:ss"));

	// 5. Deklaratywne wygenerowanie i zapis migawki (Snapshot)
	println!("🏗️");
	let snapshot = Snapshot::new()
		.title("_4GEN (CODE DUMP / SNAPSHOT) v:")
		.title_sufix_date_fmt("WYYY-WWAAA-D hh:mm:ss.zzz|tt:qq")
		.name("_4Gen_")
		.name_sufix_date_fmt("WYYY-WWAAA-Dhhmm-ssttqq")
		.output_dir("./dump")
		.max_single_file_size(524288) // 512 KiB
		.content_skip(["./assets/**", "*.svg"])
		.add_footer_summary_setups(true)
		.add_fotter_customize_text(
			"\n---\n*Generated automatically by 4gen xtask via j-Cis/{querypath,querypath-fmt,temporal-fmt} (crates.io|github.com)*\n",
		);

	match snapshot.generate_and_save(&results, &idn, &fmt) {
		Ok(path) => println!("✅ : {}", path.display()),
		Err(e) => eprintln!("❌ : {}", e),
	}
}
