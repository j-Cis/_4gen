#[derive(Clone, Copy, Default, Debug)]
pub struct SystemModifiers {

	pub alt: bool,

pub struct SystemStatus {
	pub status_msg: String,
	pub modifiers: SystemModifiers,
	pub menu_expanded: bool,
}

impl Default for SystemStatus {
	fn default() -> Self {
		Self {
			status_msg: "Gotowy".into(),
			modifiers: SystemModifiers::default(),
			menu_expanded: false,
		}
	}
}

impl SystemStatus {
	pub fn show(&mut self, ui: &mut egui::Ui) {
		// 1. CZYSTY PASEK STANU NA DOLE (Bez ciasnych guzików modyfikatorów)
		ui.horizontal(|ui| {
			ui.label(&self.status_msg);

			ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
				ui.label("_4gen · wersja rozwojowa · v:0.1.0-dev.2");
			});
		});

		// 2. NIEPRZYKRYTA SIATKA 2x2 W LEWYM DOLNYM ROGU
		self.show_floating_modifiers(ui.ctx());
	}

	fn show_floating_modifiers(&mut self, ctx: &egui::Context) {
		let screen_rect = ctx.input(|i| i.raw.screen_rect.unwrap_or(Rect::EVERYTHING));

		let cell_size = Vec2::new(58.0, 32.0);
		let gap = 5.0;
		let margin_x = 8.0;
		let margin_bottom = 28.0; // Wysokość tuż nad dolnym paskiem stanu

		// 2A: Pozycja bazowa wyzwalacza (Lewy dolny róg)
		let pos_2a = Pos2::new(
			screen_rect.min.x + margin_x,
			screen_rect.max.y - margin_bottom - cell_size.y,
		);

		// 1A: Pozycja lewego górnego rogu siatki (dla stanu rozwiniętego)
		let pos_top_left = Pos2::new(pos_2a.x, pos_2a.y - cell_size.y - gap);

		let (phys_shift, phys_ctrl, phys_alt) = ctx.input(|i| (
			i.modifiers.shift,
			i.modifiers.command || i.modifiers.ctrl,
			i.modifiers.alt,
		));

		let area_pos = if self.menu_expanded { pos_top_left } else { pos_2a };

		egui::Area::new(Id::unique("floating_system_modifiers_grid"))
			.order(Order::Tooltip) // Zawsze na samej górze - nic go nie przykryje
			.fixed_pos(area_pos)
			.show(ctx, |ui| {
				ui.style_mut().interaction.selectable_labels = false;
				ui.spacing_mut().item_spacing = Vec2::new(gap, gap);

				if self.menu_expanded {
					// --- RZĄD 1: 1A (Ctrl) i 1B (Shift) ---
					ui.horizontal(|ui| {
						let ctrl_active = self.modifiers.ctrl || phys_ctrl;
						let ctrl_btn = egui::Button::new("⌘ Ctrl").min_size(cell_size).selected(ctrl_active);
						if ui.add(ctrl_btn).on_hover_text("Wirtualny Ctrl (1A)").clicked() {
							self.modifiers.ctrl = !self.modifiers.ctrl;
						}

						let shift_active = self.modifiers.shift || phys_shift;
						let shift_btn = egui::Button::new("⇧ Shift").min_size(cell_size).selected(shift_active);
						if ui.add(shift_btn).on_hover_text("Wirtualny Shift (1B)").clicked() {
							self.modifiers.shift = !self.modifiers.shift;
						}
					});

					// --- RZĄD 2: 2A (Wyzwalacz) i 2B (Alt) ---
					ui.horizontal(|ui| {
						let trig_btn = egui::Button::new("✕ SK").min_size(cell_size).selected(true);
						if ui.add(trig_btn).on_hover_text("Zwiń przyciski modyfikatorów (2A)").clicked() {
							self.menu_expanded = false;
						}

						let alt_active = self.modifiers.alt || phys_alt;
						let alt_btn = egui::Button::new("⌥ Alt").min_size(cell_size).selected(alt_active);
						if ui.add(alt_btn).on_hover_text("Wirtualny Alt (2B)").clicked() {
							self.modifiers.alt = !self.modifiers.alt;
						}
					});
				} else {
					// --- STAN ZWINIĘTY: Tylko 2A w lewym dolnym rogu ---
					let trig_btn = egui::Button::new("⌨ SK").min_size(cell_size).selected(false);
					if ui.add(trig_btn).on_hover_text("Rozwiń skróty klawiszowe (Ctrl, Shift, Alt)").clicked() {
						self.menu_expanded = true;
					}
				}
			});
	}
}

#[derive(Clone, Copy, PartialEq)]
pub enum SideNavPlacement {

	Right,
}

#[derive(Clone)]
pub enum SideNavItem<'a> {
	Action {
		id: &'static str,
		icon: Image<'a>,
		tooltip: &'static str,
	},
	Separator,
}

pub struct SideNav<'a> {
	pub placement: SideNavPlacement,
	pub items: Vec<SideNavItem<'a>>,
}

impl<'a> SideNav<'a> {
	pub fn left(items: Vec<SideNavItem<'a>>) -> Self {
		Self {
			placement: SideNavPlacement::Left,
			items,
		}
	}

	pub fn right(items: Vec<SideNavItem<'a>>) -> Self {
		Self {
			placement: SideNavPlacement::Right,
			items,
		}
	}

	pub fn show(&self, ui: &mut Ui) -> Vec<&'static str> {
		let mut activated = Vec::new();

		ui.spacing_mut().item_spacing =
			egui::vec2(ico::ICON_PADDING_MAIN_AXIS, ico::ICON_PADDING_MAIN_AXIS);

		ui.with_layout(Layout::top_down(Align::Center), |ui| {
			for item in &self.items {
				self.show_item(ui, item, &mut activated);
			}
		});

		activated
	}

	fn show_item(&self, ui: &mut Ui, item: &SideNavItem, activated: &mut Vec<&'static str>) {
		match item {
			SideNavItem::Separator => {
				ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
				ui.separator();
				ui.add_space(ico::ICON_PADDING_MAIN_AXIS);
			}

use eframe::egui::{self, Color32, CursorIcon, Frame, Id, Key, Margin, Order, Vec2};

pub struct AutocompleteResult {
	pub committed_text: Option<String>,
}

pub fn render_autocomplete_input(
	ui: &mut egui::Ui,
	id: Id,
	buffer: &mut String,
	pool: &[String],
	allow_custom: bool,
) -> AutocompleteResult {
	let mut committed = None;

	let matches: Vec<&String> = if buffer.trim().is_empty() {
		Vec::new()
	} else {
		let needle = buffer.to_lowercase();
		pool.iter()
			.filter(|cand| cand.to_lowercase().contains(&needle))
			.take(6)
			.collect()
	};

	let input_id = id.with("text_edit");
	let input_resp = ui.add(
		egui::TextEdit::singleline(buffer)
			.id(input_id)
			.desired_width(100.0)
			.frame(Frame::NONE)
			.hint_text("Wpisz..."),
	);

	let popup_id = id.with("autocomplete_popup");
	let is_open_id = id.with("autocomplete_is_open");

	let was_open = ui.data(|d| d.get_temp::<bool>(is_open_id)).unwrap_or(false);
	let mut should_open = (input_resp.has_focus() && !matches.is_empty()) || (was_open && !matches.is_empty());

	if (input_resp.has_focus() || input_resp.lost_focus()) && ui.input(|i| i.key_pressed(Key::Enter)) {
		let trimmed = buffer.trim();
		if !trimmed.is_empty() {
			if let Some(first_match) = matches.first() {
				committed = Some((*first_match).clone());
				buffer.clear();
				should_open = false;
			} else if allow_custom {
				committed = Some(trimmed.to_string());
				buffer.clear();
				should_open = false;
			}
		}
	}

	if should_open {
		let popup_pos = input_resp.rect.left_bottom() + Vec2::new(0.0, 4.0);
		let mut clicked_item = None;

		let area_resp = egui::Area::new(popup_id)
			.fixed_pos(popup_pos)
			.order(Order::Tooltip)
			.show(ui.ctx(), |pop_ui| {
				Frame::popup(pop_ui.style())
					.fill(Color32::from_rgb(32, 34, 38))
					.inner_margin(Margin::same(4))
					.show(pop_ui, |ui| {
						ui.set_min_width(120.0);
						for candidate in &matches {
							let item_resp = ui.selectable_label(false, *candidate);
							if item_resp.hovered() {
								ui.ctx().set_cursor_icon(CursorIcon::PointingHand);
							}
							if item_resp.clicked() {
								clicked_item = Some((*candidate).clone());
							}
						}
					});
			});

		if let Some(chosen) = clicked_item {
			committed = Some(chosen);
			buffer.clear();
			should_open = false;
			ui.memory_mut(|m| m.request_focus(input_id));
		} else if ui.input(|i| i.pointer.any_pressed())
			&& let Some(pos) = ui.input(|i| i.pointer.interact_pos())
			&& !area_resp.response.rect.contains(pos)
			&& !input_resp.rect.contains(pos)
		{
			should_open = false;
		}
	}

	ui.data_mut(|d| d.insert_temp(is_open_id, should_open));

	AutocompleteResult {
		committed_text: committed,
	}
}

	pub fn show(mut self, ui: &mut egui::Ui) -> Vec<ChipAction> {
		let mut actions = Vec::new();

		let container_frame = Frame::new()
			.fill(Color32::from_rgb(28, 30, 34))
			.stroke(Stroke::new(1.0, Color32::from_gray(60)))
			.corner_radius(6.0)
			.inner_margin(Margin::same(4));

		container_frame.show(ui, |ui| {
			ui.style_mut().interaction.selectable_labels = false;
			ui.spacing_mut().item_spacing = Vec2::new(6.0, 6.0);

			match (self.mode, self.overflow) {
				(ChipContainerMode::MultiLine, ChipOverflow::Wrap) => {
					ui.horizontal_wrapped(|ui| {
						self.render_chips_flow(ui, 0, self.items.len(), &mut actions);
						self.render_input_slot(ui, &mut actions);
					});
				}

				(ChipContainerMode::SingleLine, ChipOverflow::Scroll) => {
					egui::ScrollArea::horizontal()
						.id_salt(self.id.with("chips_hscroll"))
						.auto_shrink([false, true])
						.scroll_bar_visibility(egui::scroll_area::ScrollBarVisibility::AlwaysHidden)
						.show(ui, |ui| {
							ui.horizontal(|ui| {
								self.render_chips_flow(ui, 0, self.items.len(), &mut actions);
								self.render_input_slot(ui, &mut actions);
							});
						});
				}

				(ChipContainerMode::MultiLine, ChipOverflow::Scroll) => {
					egui::ScrollArea::vertical()
						.id_salt(self.id.with("chips_vscroll"))
						.auto_shrink([false, true])
						.max_height(80.0)
						.show(ui, |ui| {
							ui.horizontal_wrapped(|ui| {
								self.render_chips_flow(ui, 0, self.items.len(), &mut actions);
								self.render_input_slot(ui, &mut actions);
							});
						});
				}

				(_, ChipOverflow::Popup { max_visible }) => {
					let total = self.items.len();
					let visible_count = max_visible.min(total);
					let hidden_count = total.saturating_sub(visible_count);

					ui.horizontal(|ui| {
						self.render_chips_flow(ui, 0, visible_count, &mut actions);

						if hidden_count > 0 {
							let popup_id = self.id.with("hidden_chips_popup");
							let is_open_id = self.id.with("hidden_chips_is_open");
							let mut is_open = ui.data(|d| d.get_temp::<bool>(is_open_id)).unwrap_or(false);

							let badge_text = format!("+{} ▾", hidden_count);
							let btn_resp = ui.selectable_label(is_open, egui::RichText::new(badge_text).strong());
							if btn_resp.hovered() {
								ui.ctx().set_cursor_icon(CursorIcon::PointingHand);
							}

							if btn_resp.clicked() {
								is_open = !is_open;
								ui.data_mut(|d| d.insert_temp(is_open_id, is_open));
							}

							if is_open {
								let popup_pos = btn_resp.rect.left_bottom() + Vec2::new(0.0, 4.0);
								let popup_resp = egui::Area::new(popup_id)
									.fixed_pos(popup_pos)
									.order(Order::Tooltip)
									.show(ui.ctx(), |pop_ui| {
										Frame::popup(pop_ui.style())
											.fill(Color32::from_rgb(32, 34, 38))
											.inner_margin(Margin::same(6))
											.show(pop_ui, |ui| {
												ui.vertical(|ui| {
													self.render_chips_flow(ui, visible_count, total, &mut actions);
												});
											});
									});

								if ui.input(|i| i.pointer.any_pressed())
									&& let Some(pos) = ui.input(|i| i.pointer.interact_pos())
									&& !popup_resp.response.rect.contains(pos)
									&& !btn_resp.rect.contains(pos)
								{
									ui.data_mut(|d| d.insert_temp(is_open_id, false));
								}
							}
						}

						self.render_input_slot(ui, &mut actions);
					});
				}

				_ => {
					ui.horizontal(|ui| {
						self.render_chips_flow(ui, 0, self.items.len(), &mut actions);
						self.render_input_slot(ui, &mut actions);
					});
				}
			}
		});

		for action in &actions {
			if let ChipAction::Removed(id) = action {
				self.items.retain(|c| c.id() != id);
			}
		}

		actions
	}

	fn render_chips_flow(
		&mut self,
		ui: &mut egui::Ui,
		start_idx: usize,
		end_idx: usize,
		actions: &mut Vec<ChipAction>,
	) {
		for idx in start_idx..end_idx {
			if idx >= self.items.len() {
				break;
			}
			let item = &mut self.items[idx];
			let is_selected = self.selected_chip.as_deref() == Some(item.id());
			let out = render_chip(ui, item, is_selected);

			if out.remove_clicked {
				actions.push(ChipAction::Removed(item.id().to_string()));
			} else if out.add_param_clicked {
				item.convert_to_parametric();
				actions.push(ChipAction::Clicked(item.id().to_string()));
			} else if out.param_changed {
				if let Some(val) = item.param_value() {
					actions.push(ChipAction::ParamChanged {
						id: item.id().to_string(),
						display_value: val,
					});
				}
			} else if out.clicked {
				*self.selected_chip = Some(item.id().to_string());
				actions.push(ChipAction::Clicked(item.id().to_string()));
			}
		}
	}

	fn render_input_slot(&mut self, ui: &mut egui::Ui, actions: &mut Vec<ChipAction>) {
		let res = render_autocomplete_input(
			ui,
			self.id.with("autocomplete"),
			self.input_buffer,
			self.autocomplete_pool,
			self.limits.allow_custom,
		);

		if let Some(text) = res.committed_text {
			let id = format!("chip_{}", ui.input(|i| i.time));

			let new_chip = if let Some((key, val)) = text.split_once(':') {
				let k = key.trim();
				let v = val.trim();

				let slot = match k {
					"Płeć" => ChipSlot::select(
						if v.is_empty() { "Kobieta" } else { v },
						vec!["Kobieta".into(), "Mężczyzna".into()],
					),
					"Wiek" => ChipSlot::range(18.0, 65.0, 0.0, 120.0),
					_ => ChipSlot::text(v),
				};

				ChipItem::Parametric {
					id,
					key: k.to_string(),
					slot,
					is_editing: v.is_empty(),
				}
			} else {
				ChipItem::Simple {
					id,
					label: text.clone(),
				}
			};

			if self.limits.can_add(new_chip.key_or_label(), self.items) {
				actions.push(ChipAction::Added(new_chip.id().to_string()));
				self.items.push(new_chip);
			}
		}
	}
}

				return;
			}

			return;

			if let Some(idx) = existing_idx {
				self.sort_rules[idx].1 = match self.sort_rules[idx].1 {
					SortDir::Asc => SortDir::Desc,
					SortDir::Desc => SortDir::Asc,

			} else {
				self.sort_rules.push((col_id.to_string(), SortDir::Asc));
			}

					match self.sort_rules[idx].1 {
						SortDir::Asc => SortDir::Desc,
						SortDir::Desc => SortDir::Asc,
					}
				} else {
					SortDir::Asc
				}
			} else {
				SortDir::Asc
			};
			self.sort_rules.clear();
			self.sort_rules.push((col_id.to_string(), new_dir));
		}
	}

		if !config.allow_selection {
			return;
		}

		if ctrl_pressed && config.multi_select {
			if self.selected_rows.contains(&row_idx) {
				self.selected_rows.remove(&row_idx);
			} else {
				if let Some(limit) = config.max_selection
					&& self.selected_rows.len() >= limit {
					return;
				}
				self.selected_rows.insert(row_idx);
			}
		} else {
			self.selected_rows.clear();
			self.selected_rows.insert(row_idx);
		}
	}

						ui.spacing_mut().item_spacing.x = 8.0;

						for item in left_items {
							render_item(ui, *item, row_idx, list_len, col_id, state, &mut swap_action);
						}

						if !right_items.is_empty() {
							ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |r_ui| {
								r_ui.spacing_mut().item_spacing.x = 8.0;

								for item in right_items.iter().rev() {
									render_item(r_ui, *item, row_idx, list_len, col_id, state, &mut swap_action);
								}

								r_ui.with_layout(egui::Layout::left_to_right(egui::Align::Center), |t_ui| {
									handle_response = Some(render_title_handle(t_ui, item_id, row_idx, col_id, state));
								});
							});
						} else {
							handle_response = Some(render_title_handle(ui, item_id, row_idx, col_id, state));
						}
					});

					// Wizualizacja przeciągania DND na klocku tytułowym
					if let Some(hr) = handle_response
						&& let (Some(pointer), Some(hovered_payload)) = (
							ui.input(|i| i.pointer.interact_pos()),
							hr.dnd_hover_payload::<usize>(),
						) {
						let rect = hr.rect;
						let stroke = Stroke::new(2.0, Color32::from_rgb(0, 120, 215));

						let insert_target = if *hovered_payload == row_idx {
							ui.painter().hline(rect.x_range(), rect.center().y, stroke);
							row_idx
						} else if pointer.y < rect.center().y {
							ui.painter().hline(rect.x_range(), rect.top(), stroke);
							row_idx
						} else {
							ui.painter().hline(rect.x_range(), rect.bottom(), stroke);
							row_idx + 1
						};

						if let Some(dragged_payload) = hr.dnd_release_payload() {
							from_idx = Some(*dragged_payload);
							to_idx = Some(insert_target);
						}
					}

					ui.separator();
				}
			});

			if let Some((a, b)) = swap_action {
				state.display_order.swap(a, b);
			}

			if let (Some(from), Some(to)) = (from_idx, to_idx) {
				engine::reorder_columns(state, from, to);
			}
		});
	}
}

fn render_title_handle(ui: &mut egui::Ui, item_id: Id, row_idx: usize, col_id: &str, state: &TableState) -> egui::Response {

#![allow(clippy::excessive_precision, clippy::approx_constant)]

use egui::Color32;

/// Rozszerzenie Color32 o przestrzenie barw OKLab i OKLCH.
pub trait Color32OklabExt {
	fn from_oklab(l: f32, a: f32, b: f32) -> Self;
	fn from_oklch(l: f32, c: f32, h: f32) -> Self;
}

impl Color32OklabExt for Color32 {
	fn from_oklab(l: f32, a: f32, b: f32) -> Self {
		oklab(l, a, b)
	}

	fn from_oklch(l: f32, c: f32, h: f32) -> Self {
		oklch(l, c, h)
	}
}

/// Helper: Konwersja Oklab do liniowego sRGB (zwraca r, g, b bez obcinania)
const fn oklab_to_linear_srgb(l: f32, a: f32, b: f32) -> (f32, f32, f32) {
	let l_ = l + 0.3963377774 * a + 0.2158037573 * b;
	let m_ = l - 0.1055613458 * a - 0.0638541728 * b;
	let s_ = l - 0.0894841775 * a - 1.2914855480 * b;

	let ll = l_ * l_ * l_;
	let mm = m_ * m_ * m_;
	let ss = s_ * s_ * s_;

	let r = 4.0767416621 * ll - 3.3077115913 * mm + 0.2309699292 * ss;
	let g = -1.2684380046 * ll + 2.6097574011 * mm - 0.3413193965 * ss;
	let b = -0.0041960863 * ll - 0.7034186147 * mm + 1.7076147010 * ss;

	(r, g, b)
}

/// Helper: Sprawdza czy składowe RGB mieszczą się w gamucie sRGB (0.0 - 1.0)
const fn is_in_srgb_gamut(r: f32, g: f32, b: f32) -> bool {
	r >= 0.0 && r <= 1.0 && g >= 0.0 && g <= 1.0 && b >= 0.0 && b <= 1.0
}

/// Konwersja OKLab do sRGB — dostępna w czasie kompilacji.
pub const fn oklab(l: f32, a: f32, b: f32) -> Color32 {
	if l <= 0.0 {
		return Color32::BLACK;
	}
	let (r, g, b) = oklab_to_linear_srgb(l, a, b);
	Color32::from_rgb(channel(r), channel(g), channel(b))
}

/// Konwersja OKLCH (L, C, h w stopniach) do sRGB — const.
pub const fn oklch(l: f32, c: f32, h: f32) -> Color32 {
	if l <= 0.0 {
		return Color32::BLACK;
	}
	let l_clamped = if l < 0.0 {
		0.0
	} else if l > 1.0 {
		1.0
	} else {
		l
	};
	if c <= 0.0 {
		return oklab(l, 0.0, 0.0);
	}
	let (sin_h, cos_h) = sin_cos_degrees(h);

	let (mut r, mut g, mut b) = oklab_to_linear_srgb(l_clamped, c * cos_h, c * sin_h);

	if !is_in_srgb_gamut(r, g, b) {
		let mut min_c = 0.0;
		let mut max_c = c;
		let mut iters = 0;

		let (mut best_r, mut best_g, mut best_b) = oklab_to_linear_srgb(l_clamped, 0.0, 0.0);

		while iters < 16 {
			let mid_c = min_c + (max_c - min_c) / 2.0;
			let (tr, tg, tb) = oklab_to_linear_srgb(l_clamped, mid_c * cos_h, mid_c * sin_h);

			if is_in_srgb_gamut(tr, tg, tb) {
				best_r = tr;
				best_g = tg;
				best_b = tb;
				min_c = mid_c;
			} else {
				max_c = mid_c;
			}
			iters += 1;
		}

		r = best_r;
		g = best_g;
		b = best_b;
	}

	Color32::from_rgb(channel(r), channel(g), channel(b))
}

const fn channel(linear: f32) -> u8 {
	let x = if linear < 0.0 {
		0.0
	} else if linear > 1.0 {
		1.0
	} else {
		linear
	};

	let srgb = if x <= 0.0031308 {
		12.92 * x
	} else {
		1.055 * pow_five_twelths(x) - 0.055
	};

	(srgb * 255.0 + 0.5) as u8
}

const fn pow_five_twelths(x: f32) -> f32 {
	let mut y = 1.0_f32;
	let mut i = 0;

	while i < 16 {
		let y2 = y * y;
		let y4 = y2 * y2;
		let y8 = y4 * y4;
		let y11 = y8 * y2 * y;
		let y12 = y11 * y;

		y = y - (y12 - x) / (12.0 * y11);
		i += 1;
	}

	let y2 = y * y;
	let y4 = y2 * y2;
	y4 * y
}

const fn sin_cos_degrees(degrees: f32) -> (f32, f32) {
	let mut angle = degrees;

	angle %= 360.0;
	if angle < 0.0 {
		angle += 360.0;
	}
	if angle > 180.0 {
		angle -= 360.0;
	}

	let mut x = angle * 0.017453292519943295;
	let mut cos_sign = 1.0_f32;

	if x > 1.5707963267948966 {
		x = 3.141592653589793 - x;
		cos_sign = -1.0;
	} else if x < -1.5707963267948966 {
		x = -3.141592653589793 - x;
		cos_sign = -1.0;
	}

	let x2 = x * x;

	let sin = x
		* (1.0
			+ x2 * (-1.0 / 6.0
				+ x2 * (1.0 / 120.0
					+ x2 * (-1.0 / 5040.0
						+ x2 * (1.0 / 362880.0
							+ x2 * (-1.0 / 39916800.0 + x2 * (1.0 / 6227020800.0)))))));
	let cos = 1.0
		+ x2 * (-1.0 / 2.0
			+ x2 * (1.0 / 24.0
				+ x2 * (-1.0 / 720.0
					+ x2 * (1.0 / 40320.0
						+ x2 * (-1.0 / 3628800.0
							+ x2 * (1.0 / 479001600.0 + x2 * (-1.0 / 87178291200.0)))))));

	(sin, cos * cos_sign)
}

	pub fn show(&mut self, ui: &mut egui::Ui, sys: &mut SystemStatus) {
		let mut row_indices: Vec<usize> = (0..self.records.len()).collect();

		let sort_rules = self.state.sort_rules.clone();
		if !sort_rules.is_empty() {
			row_indices.sort_by(|&a_idx, &b_idx| {
				let rec_a = &self.records[a_idx];
				let rec_b = &self.records[b_idx];

				for (col_id, dir) in &sort_rules {
					let cmp = match col_id.as_str() {
						"id" => rec_a.id.cmp(&rec_b.id),
						"first_names" => rec_a.first_names.cmp(&rec_b.first_names),
						"last_names" => rec_a.last_names.cmp(&rec_b.last_names),
						"gender" => rec_a.gender.cmp(&rec_b.gender),
						"birth_date" => rec_a.birth_date.cmp(&rec_b.birth_date),
						"birth_place" => rec_a.birth_place.cmp(&rec_b.birth_place),
						_ => Ordering::Equal,
					};

					if cmp != Ordering::Equal {
						return if *dir == SortDir::Asc { cmp } else { cmp.reverse() };
					}
				}
				Ordering::Equal
			});
		}

		let mut table = AdvancedTable {
			theme: &self.theme,
			config: &self.config,
			state: &mut self.state,
		};

		let action = table.show(ui, &row_indices, &sys.modifiers, |true_idx, col_id, cell_ui| {
			let item = &self.records[true_idx];
			match col_id {
				"id" => { cell_ui.label(item.id.to_string()); }
				"first_names" => { cell_ui.label(&item.first_names); }
				"last_names" => { cell_ui.label(&item.last_names); }
				"gender" => { cell_ui.label(&item.gender); }
				"birth_date" => { cell_ui.label(item.birth_date.as_deref().unwrap_or("-")); }
				"birth_place" => { cell_ui.label(item.birth_place.as_deref().unwrap_or("-")); }
				"tags" => { cell_ui.label(item.tags.join(", ")); }
				_ => {}
			}
		});

		if let Some(TableAction::ConfigColumns) = action {
			self.config_window.window.open();
		}

		self.config_window.show(ui, &mut self.state);
	}
